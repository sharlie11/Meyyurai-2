# Readme Meyyurai

Meyyurai is an AI-powered human guidance, rights protection, emergency assistance, and legal navigation platform.

This repository contains a production-oriented first build:

- FastAPI backend
- Searchable structured knowledge database
- Responsive PWA frontend
- Guided intake flow with follow-up questions
- Emergency assistance mode
- Dashboard screens
- Multilingual UI support for English, Tamil, Hindi, Telugu, and Malayalam

## Run Locally

```powershell
python -m venv .venv
.\.venv\Scripts\Activate.ps1
pip install -r requirements.txt
uvicorn app.main:app --reload
```

Open `http://127.0.0.1:8000`.

## Important Legal Safety

The sample knowledge database is intentionally small and must be reviewed by qualified legal professionals before real-world deployment. The app is designed to retrieve from verified records and show source references rather than invent laws.


#main.py
from __future__ import annotations

from pathlib import Path
from typing import Any

from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from fastapi.responses import FileResponse
from fastapi.staticfiles import StaticFiles
from pydantic import BaseModel, Field

from app.services.guidance import GuidanceEngine
from app.services.knowledge_base import KnowledgeBase

BASE_DIR = Path(__file__).resolve().parent.parent
STATIC_DIR = BASE_DIR / "web"
DATA_DIR = BASE_DIR / "data"

app = FastAPI(
    title="Meyyurai API",
    description="Human guidance, rights protection, emergency assistance, and legal navigation platform.",
    version="0.1.0",
)

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

knowledge_base = KnowledgeBase(DATA_DIR / "knowledge_base.json")
guidance_engine = GuidanceEngine(knowledge_base)


class GuidanceRequest(BaseModel):
    description: str = Field(..., min_length=4)
    language: str | None = None
    answers: dict[str, str] = Field(default_factory=dict)
    emergency: bool = False


class SearchRequest(BaseModel):
    query: str = Field(..., min_length=2)
    language: str | None = None
    category: str | None = None


@app.get("/api/health")
def health() -> dict[str, str]:
    return {"status": "ok", "service": "meyyurai"}


@app.post("/api/guidance")
def create_guidance(request: GuidanceRequest) -> dict[str, Any]:
    return guidance_engine.build_guidance(
        description=request.description,
        language=request.language,
        answers=request.answers,
        emergency=request.emergency,
    )


@app.post("/api/search")
def search(request: SearchRequest) -> dict[str, Any]:
    results = knowledge_base.search(
        query=request.query,
        category=request.category,
        limit=8,
    )
    return {"query": request.query, "results": results}


@app.get("/api/resources/{resource_id}")
def get_resource(resource_id: str) -> dict[str, Any]:
    resource = knowledge_base.get(resource_id)
    if not resource:
        raise HTTPException(status_code=404, detail="Resource not found")
    return resource


app.mount("/assets", StaticFiles(directory=STATIC_DIR / "assets"), name="assets")


@app.get("/manifest.webmanifest")
def manifest() -> FileResponse:
    return FileResponse(STATIC_DIR / "manifest.webmanifest", media_type="application/manifest+json")


@app.get("/service-worker.js")
def service_worker() -> FileResponse:
    return FileResponse(STATIC_DIR / "service-worker.js", media_type="application/javascript")


@app.get("/{path:path}")
def spa(path: str) -> FileResponse:
    file_path = STATIC_DIR / path
    if path and file_path.exists() and file_path.is_file():
        return FileResponse(file_path)
    return FileResponse(STATIC_DIR / "index.html")
